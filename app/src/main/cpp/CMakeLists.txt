cmake_minimum_required(VERSION 3.22.1)
project(echoguide_hrtf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/steamaudio")
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(STEAM_ABI android-armv8)
elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(STEAM_ABI android-armv7)
elseif(ANDROID_ABI STREQUAL "x86_64")
    set(STEAM_ABI android-x64)
else()
    message(FATAL_ERROR "Unsupported ABI ${ANDROID_ABI}")
endif()
add_library(phonon SHARED IMPORTED)
set_target_properties(phonon PROPERTIES IMPORTED_LOCATION "${SDK}/lib/${STEAM_ABI}/libphonon.so")
add_library(echoguide_hrtf SHARED steam_audio.cpp)
target_include_directories(echoguide_hrtf PRIVATE "${SDK}/include")
target_link_libraries(echoguide_hrtf PRIVATE phonon)
